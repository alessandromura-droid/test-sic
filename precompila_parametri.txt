# Parametri di precompila_v2 (build_routing_graph_v2.py).

# Diametro massimo (metri) di un gruppo di nodi fusi insieme (passo
# B del clustering) - un incrocio reale piu' largo di cosi' resta
# su piu' nodi distinti invece di diventarne uno solo.
node_cluster_max_diameter_m = 30.0

# Oltre questo numero di incroci (passo C), una coppia di archi e'
# considerata doppione/parallelismo spurio e scartata per intero.
max_crossings_per_pair = 2

# Distanza massima (metri) fra l'estremita' "a vuoto" di un arco e
# il tracciato di un altro arco perche' il passo D li colleghi
# (caso tipico: stesso incrocio reale digitalizzato da due fonti
# con un piccolo scarto).
dangle_tolerance_m = 50.0

# Soglia (gradi) oltre la quale un vertice e' un'inversione
# "avanti e indietro" da rimuovere (angolo fra vettore di entrata
# e di uscita).
uturn_angle_threshold_deg = 160.0

# Dimensione (pixel) del lato del canvas quadrato di ogni JPEG di
# debug per sorgente.
debug_image_size_px = 8000

# Spessore (pixel) delle linee degli archi nelle JPEG di debug.
debug_image_edge_width_px = 3

# Raggio (pixel) dei pallini rossi dei nodi nelle JPEG di debug.
debug_image_node_radius_px = 2.0

# Latitudine del centro dell'area fissa inquadrata nelle JPEG di
# debug (default: Piazza Venezia, "Roma km 0").
debug_image_center_lat = 41.896435

# Longitudine del centro dell'area fissa inquadrata nelle JPEG di
# debug.
debug_image_center_lon = 12.482531

# Raggio (km) dell'area inquadrata nelle JPEG di debug - meta' del
# lato del quadrato (20 = area 40x40km). La risoluzione (m/px) si
# ricalcola da sola da questo raggio e da debug_image_size_px.
debug_image_radius_km = 20.0

# Latitudine del centro dell'area in cui RouteCoherenceTest sceglie
# partenza e arrivo dei percorsi di test (default: Piazza Venezia).

# Longitudine del centro dell'area in cui RouteCoherenceTest sceglie
# partenza e arrivo dei percorsi di test.
debug_route_test_center_lon = 12.482531

# Raggio (km) del cerchio, attorno al centro sopra, dentro cui devono
# stare partenza e arrivo dei percorsi di RouteCoherenceTest (il
# percorso in mezzo puo' uscire). Letto solo dal test Kotlin.
debug_route_test_radius_km = 20.0

# Colore delle linee degli archi nelle JPEG di debug, 'r,g,b' (0-255
# ciascuno) - libero, per distinguerli bene dallo sfondo OSM.

# Colore dei pallini dei nodi nelle JPEG di debug, 'r,g,b'.

# recente dello script (non toccano le righe sopra) ---

# Tre modalita': 'disable' (default) solo il nostro loop A/B/B2/C/D;
# 'enable' solo neatnet + una passata del nostro codice come rete di
# sicurezza; 'both' neatnet PRIMA, poi il loop completo (non una sola
# passata) - vedi il docstring del modulo per i dettagli.

# Tetto di sicurezza al numero di iterazioni del loop A/B/B2/C/D per-
# sorgente quando use_neatnet e' false - la convergenza non e'
# garantita, quindi il loop si ferma comunque qui se non si stabilizza.

# --- parametri aggiunti automaticamente da una versione piu'
# recente dello script (non toccano le righe sopra) ---
